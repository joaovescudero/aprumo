<objective>
Pre-create cluster-global Postgres roles (`aprumo_app`, `aprumo_migration`) once in `globalSetup.ts`, immediately after `container.start()` resolves and before Vitest forks any worker. This eliminates the non-atomic race in `0001_roles.sql` that causes `PostgresError 23505` when parallel test files call `applyMigrationsToSchema` against a fresh container.

Purpose: The `IF NOT EXISTS` guard in `0001_roles.sql` is not atomic across sessions — two forks can both observe the role missing, then both attempt `CREATE ROLE`, with the second hitting a unique violation on `pg_authid_rolname_index`. By pre-creating roles in the single-threaded `globalSetup`, every subsequent `applyMigrationsToSchema` invocation finds the roles already present and the `IF NOT EXISTS` check short-circuits harmlessly. Migration files are not modified; the production migrate path is not touched.

Output:
- `packages/core/tests/helpers/globalSetup.race.test.ts` — failing test (RED) that proves 23505 surfaces without pre-creation
- `packages/core/tests/globalSetup.ts` — role pre-creation added after container start (GREEN)
</objective>

import { PostgreSqlContainer, type StartedPostgreSqlContainer } from "@testcontainers/postgresql";
import { PG_IMAGE } from "./setup/container.js";

let container: StartedPostgreSqlContainer | undefined;

export async function setup(project: { provide: (key: string, value: unknown) => void }) {
  const useReuse = process.env.APRUMO_TEST_REUSE === "1";
  let builder = new PostgreSqlContainer(PG_IMAGE);
  if (useReuse) {
    builder = builder.withReuse();
  }
  try {
    container = await builder.start();
    project.provide("pgUri", container.getConnectionUri());
  } catch (err) {
    const msg = err instanceof Error ? err.message : String(err);
    console.warn(`[globalSetup] Docker unavailable — integration tests will be skipped: ${msg}`);
    project.provide("pgUri", "");
  }
}

export async function teardown() {
  if (container) {
    await container.stop();
  }
}

// Role/type duplicates (23505 / 42710): handled via savepoints so the outer
// transaction stays alive when a concurrent fork has already created a global role.
await tx.savepoint(async (sp) => {
  try {
    await sp.unsafe(trimmed);
  } catch (err) {
    const pgCode = err instanceof Error && "code" in err ? (err as { code: string }).code : undefined;
    const firstToken = trimmed.replace(/--[^\n]*/g, "").trim().toUpperCase().slice(0, 20);
    const isRoleStatement = firstToken.startsWith("CREATE ROLE") || firstToken.startsWith("DO ");
    if ((pgCode === "23505" || pgCode === "42710") && isRoleStatement) {
      return; // Role already created by a concurrent fork — safely idempotent.
    }
    throw err;
  }
});

    //

    // EXCEPTION duplicate_object — the atomic, correct idempotency mechanism.
    //
    // Per D-40 reuse mode (.withReuse): the warm container already has the roles;
    // EXCEPTION duplicate_object silently absorbs 42710 — idempotent.

    // Production migrate.ts is NOT affected — 0001_roles.sql still runs there
